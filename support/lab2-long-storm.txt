# A stream read may be short, so bufstorm can finish before a whole line.
# Subsequent characters may become Shell commands; hi must still run afterward.

EXPECT BUFSTORM lines=4 bytes=[1-9][0-9]*
WAIT 300
SEND
WAIT 100
SEND hi
EXPECT hi: user program running, pid=[0-9]+
